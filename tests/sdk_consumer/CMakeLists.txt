cmake_minimum_required(VERSION 3.24)
project(EcoSphereExternalConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
foreach(name IN ITEMS SDK_ROOT CORE_LIBRARY SCENE_LIBRARY)
    if(NOT DEFINED ${name})
        message(FATAL_ERROR "Missing declared dependency: ${name}")
    endif()
endforeach()
add_library(EcoSphereSDK INTERFACE)
target_include_directories(EcoSphereSDK INTERFACE "${SDK_ROOT}/include")
target_include_directories(EcoSphereSDK SYSTEM INTERFACE "${SDK_ROOT}/third_party/glm")
target_link_libraries(EcoSphereSDK INTERFACE "${CORE_LIBRARY}" "${SCENE_LIBRARY}")
add_executable(sdk_consumer main.cpp LessonModule.cpp)
target_link_libraries(sdk_consumer PRIVATE EcoSphereSDK)
if(NOT MSVC)
    target_compile_options(sdk_consumer PRIVATE -Wall -Wextra -Wpedantic)
endif()
